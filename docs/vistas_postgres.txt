    tm.accion_movimiento,
    tm.genera_gmf,
    e.modulo,
    m.descripcion_modulo AS nombre_modulo,
    e.tarjeta,

     JOIN depositos.cuentas_ahorro c ON ((c.id_cuenta_ahorro = e.id_cuenta_ahorro)))
     JOIN hoja_vida.datos_personales dp ON ((dp.id_datos_personal = c.id_datos_personal)))
