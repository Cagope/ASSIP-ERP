  v_id_agencia        integer;
  v_codigo_cuenta     text;

  v_naturaleza        text;
  v_nivel             integer;

  END IF;

    EXCEPTION WHEN others THEN
      v_nivel := NULL;

    id_tipo_especial_cuenta,
    control_entrada_salida,

    fecha_creacion,

hoja_vida,pg_guardar_datos_personales_json,"CREATE OR REPLACE FUNCTION hoja_vida.pg_guardar_datos_personales_json(p_data jsonb, p_id integer DEFAULT NULL::integer)
 RETURNS integer
 LANGUAGE plpgsql
AS $function$
DECLARE
  -- Tabla destino FIJA
  p_target CONSTANT regclass := 'hoja_vida.datos_personales'::regclass;

  v_sql  text;
  v_id   integer;
  pkcol  name;

  ins_cols text;
  ins_vals text;
  upd_set  text;

  has_conflict_cols boolean;
  json_keys  text;
  tdoc text := p_data->>'tipo_documento';
  doc  text := p_data->>'documento';

  -- control de timestamps
  has_fc  boolean;  -- existe columna fecha_creacion
  has_fe  boolean;  -- existe columna fecha_edicion
  fc_typ  text;     -- tipo real de fecha_creacion
  fe_typ  text;     -- tipo real de fecha_edicion

  _sep text;
BEGIN
  -- PK de la tabla (busca la PK real; si no la encuentra, intenta nombres típicos)
  SELECT a.attname INTO pkcol
  FROM pg_index i
  JOIN pg_attribute a ON a.attrelid = i.indrelid AND a.attnum = ANY(i.indkey)
  WHERE i.indrelid = p_target AND i.indisprimary
  LIMIT 1;

  IF pkcol IS NULL THEN
    SELECT a.attname INTO pkcol
    FROM pg_attribute a
    WHERE a.attrelid = p_target
      AND a.attnum > 0 AND NOT a.attisdropped
      AND a.attname IN ('id_datos_personal','id_datos_personales','id_persona','id')
    LIMIT 1;
  END IF;

  IF pkcol IS NULL THEN
    RAISE EXCEPTION 'No se pudo determinar la PK de %', p_target::text;
  END IF;

  -- columnas de timestamps presentes y sus tipos
  SELECT EXISTS (
           SELECT 1 FROM pg_attribute
           WHERE attrelid = p_target AND attname = 'fecha_creacion'
                 AND attnum > 0 AND NOT attisdropped
         ),
         EXISTS (
           SELECT 1 FROM pg_attribute
           WHERE attrelid = p_target AND attname = 'fecha_edicion'
                 AND attnum > 0 AND NOT attisdropped
         )
    INTO has_fc, has_fe;

  IF has_fc THEN
    SELECT format_type(a.atttypid, a.atttypmod)
      INTO fc_typ
    FROM pg_attribute a
    WHERE a.attrelid = p_target
      AND a.attname = 'fecha_creacion'
      AND a.attnum > 0 AND NOT a.attisdropped
    LIMIT 1;
  END IF;

  IF has_fe THEN
    SELECT format_type(a.atttypid, a.atttypmod)
      INTO fe_typ
    FROM pg_attribute a
    WHERE a.attrelid = p_target
      AND a.attname = 'fecha_edicion'
      AND a.attnum > 0 AND NOT a.attisdropped
    LIMIT 1;
  END IF;

  -- Claves del JSON (diagnóstico)
  SELECT string_agg(k, ', ' ORDER BY k) INTO json_keys
  FROM jsonb_object_keys(p_data) AS k;

  -- Si viene PK explícita, UPDATE parcial con solo campos presentes en el JSON
  IF p_id IS NOT NULL THEN
    WITH cols AS (
      SELECT a.attname AS col,
             format('(NULLIF($1::jsonb->>%L, ''''))::%s',
                    a.attname, format_type(a.atttypid, a.atttypmod)) AS cast_expr
      FROM pg_attribute a
      WHERE a.attrelid = p_target
        AND a.attnum > 0 AND NOT a.attisdropped
        AND a.attname <> pkcol
        AND (p_data ? a.attname)
    )
    SELECT string_agg(format('%I = %s', col, cast_expr), ',')
      INTO upd_set
    FROM cols;

    -- Garantiza fecha_edicion en UPDATE por PK
    IF has_fe THEN
      IF upd_set IS NULL OR upd_set = '' THEN
        IF (p_data ? 'fecha_edicion') THEN
          upd_set := format('fecha_edicion = (NULLIF($1::jsonb->>%L, ''''))::%s',
                            'fecha_edicion', fe_typ);
        ELSE
          upd_set := format('fecha_edicion = now()::%s', fe_typ);
        END IF;
      ELSE
        IF (p_data ? 'fecha_edicion') THEN
          upd_set := upd_set || ', ' ||
                     format('fecha_edicion = (NULLIF($1::jsonb->>%L, ''''))::%s',
                            'fecha_edicion', fe_typ);
        ELSE
          upd_set := upd_set || ', ' || format('fecha_edicion = now()::%s', fe_typ);
        END IF;
      END IF;
    END IF;

    IF upd_set IS NULL OR upd_set = '' THEN
      RAISE EXCEPTION 'El JSON no contiene columnas válidas para %', p_target::text
      USING DETAIL = 'Claves JSON: '||COALESCE(json_keys,'(vacío)');
    END IF;

    v_sql := 'UPDATE '|| p_target::text ||' SET '|| upd_set ||
             ' WHERE '|| quote_ident(pkcol) ||' = $2 ' ||
             ' RETURNING '|| quote_ident(pkcol);

    EXECUTE v_sql USING p_data, p_id INTO v_id;

    IF v_id IS NULL THEN
      RAISE EXCEPTION 'UPDATE no devolvió ID'
      USING DETAIL = v_sql, HINT = 'Claves JSON: '||COALESCE(json_keys,'(vacío)');
    END IF;

    RETURN v_id;
  END IF;

  -- UPSERT por (tipo_documento, documento)
  SELECT (SELECT COUNT(*) FROM pg_attribute
          WHERE attrelid = p_target AND attname IN ('tipo_documento','documento')) = 2
    INTO has_conflict_cols;

  IF NOT has_conflict_cols THEN
    RAISE EXCEPTION 'La tabla % no tiene (tipo_documento, documento) para upsert', p_target::text;
  END IF;
  IF tdoc IS NULL OR tdoc = '' OR doc IS NULL OR doc = '' THEN
    RAISE EXCEPTION 'El JSON debe incluir tipo_documento y documento para el upsert'
    USING DETAIL = 'Claves JSON: '||COALESCE(json_keys,'(vacío)');
  END IF;

  WITH cols AS (
    SELECT a.attname AS col,
           format('(NULLIF($1::jsonb->>%L, ''''))::%s',
                  a.attname, format_type(a.atttypid, a.atttypmod)) AS cast_expr
    FROM pg_attribute a
    WHERE a.attrelid = p_target
      AND a.attnum > 0 AND NOT a.attisdropped
      AND a.attname <> pkcol
      AND (p_data ? a.attname)
  ),
  cols_ins AS (
    SELECT col, cast_expr FROM cols
  ),
  cols_upd AS (
    SELECT col FROM cols
    WHERE col NOT IN ('tipo_documento','documento', pkcol)
  )
  SELECT 
    (SELECT string_agg(quote_ident(ci.col), ',') FROM cols_ins ci),
    (SELECT string_agg(ci.cast_expr, ',')        FROM cols_ins ci),
    (SELECT string_agg(format('%I = EXCLUDED.%I', cu.col, cu.col), ',') FROM cols_upd cu)
  INTO ins_cols, ins_vals, upd_set;

  -- Agrega fechas si faltan
  IF ins_cols IS NULL THEN ins_cols := ''; END IF;
  IF ins_vals IS NULL THEN ins_vals := ''; END IF;

  IF has_fc AND NOT (p_data ? 'fecha_creacion') THEN
    _sep := CASE WHEN ins_cols = '' THEN '' ELSE ',' END;
    ins_cols := ins_cols || _sep || 'fecha_creacion';
    ins_vals := ins_vals || _sep || format('now()::%s', fc_typ);
  END IF;

  IF has_fe AND NOT (p_data ? 'fecha_edicion') THEN
    _sep := CASE WHEN ins_cols = '' THEN '' ELSE ',' END;
    ins_cols := ins_cols || _sep || 'fecha_edicion';
    ins_vals := ins_vals || _sep || format('now()::%s', fe_typ);
  END IF;

  IF ins_cols = '' OR ins_vals = '' THEN
    RAISE EXCEPTION 'El JSON no contiene columnas válidas para %', p_target::text
    USING DETAIL = 'Claves JSON: '||COALESCE(json_keys,'(vacío)');
  END IF;

  -- Actualiza fecha_edicion al hacer upsert
  IF has_fe THEN
    IF upd_set IS NULL OR upd_set = '' THEN
      upd_set := format('fecha_edicion = now()::%s', fe_typ);
    ELSE
      upd_set := upd_set || ', ' || format('fecha_edicion = now()::%s', fe_typ);
    END IF;
  END IF;

  IF upd_set IS NULL OR upd_set = '' THEN
    upd_set := quote_ident(pkcol)||'='||quote_ident(pkcol);
  END IF;

  v_sql := 'INSERT INTO '|| p_target::text ||' ('|| ins_cols ||') VALUES ('|| ins_vals ||') ' ||
           'ON CONFLICT (tipo_documento, documento) DO UPDATE SET ' || upd_set ||
           ' RETURNING '|| quote_ident(pkcol);

  EXECUTE v_sql USING p_data INTO v_id;

  -- Fallback
  IF v_id IS NULL THEN
    EXECUTE format(
      'SELECT %I FROM %s WHERE tipo_documento = $1 AND documento = $2 LIMIT 1',
      pkcol, p_target::text
    ) USING tdoc, doc INTO v_id;
  END IF;

  IF v_id IS NULL THEN
    RAISE EXCEPTION 'UPSERT no devolvió ID ni fallback encontró fila'
    USING DETAIL = v_sql,
          HINT = 'Verifique PK/DEFAULT e índice único (tipo_documento,documento).';
  END IF;

  RETURN v_id;
END;
$function$

    v_cel1 := NULLIF(regexp_replace(coalesce(j->>'celular_uno',''), '\D', '', 'g'), '');
    IF v_cel1 IS NOT NULL AND length(v_cel1) <> 10 THEN v_cel1 := NULL; END IF;

    v_cel2 := NULLIF(regexp_replace(coalesce(j->>'celular_dos',''), '\D', '', 'g'), '');
    IF v_cel2 IS NOT NULL AND length(v_cel2) <> 10 THEN v_cel2 := NULL; END IF;

    -- 🔹 Correo
    v_correo := left(NULLIF(btrim(j->>'correo'), ''), 100);

    -- 🔹 Catálogos
    v_id_pais         := NULLIF(j->>'id_pais', '')::int;
    v_id_departamento := NULLIF(j->>'id_departamento', '')::int;
    v_id_ciudad       := NULLIF(j->>'id_ciudad', '')::int;
    v_id_sub_zona     := NULLIF(j->>'id_sub_zona', '')::int;
    v_id_zona         := NULLIF(j->>'id_zona', '')::int;  -- opcional

    -- 🔹 Coherencia sub-zona ↔ zona
    IF v_id_sub_zona IS NOT NULL THEN
        SELECT sz.id_zona
          INTO v_id_zona_from_sub
          FROM general.sub_zonas sz
         WHERE sz.id_sub_zona = v_id_sub_zona;

        IF v_id_zona_from_sub IS NULL THEN
            RAISE EXCEPTION 'id_sub_zona % no existe en general.sub_zonas', v_id_sub_zona;
        END IF;

        IF v_id_zona IS NOT NULL AND v_id_zona <> v_id_zona_from_sub THEN
            RAISE EXCEPTION 'Inconsistencia: sub_zona % pertenece a zona %, pero llegó id_zona %',
                v_id_sub_zona, v_id_zona_from_sub, v_id_zona;
        END IF;
    END IF;

    -- 🔹 UPSERT (1:1 con persona)
    INSERT INTO hoja_vida.ubicaciones (
    id_datos_personal,
    direccion, barrio, telefono, celular_uno, celular_dos, correo,
    id_pais, id_departamento, id_ciudad,
    id_sub_zona, id_zona,
    fk_seguridad_creacion, fecha_creacion,
    fk_seguridad_edicion, fecha_edicion
	)
	VALUES (
	    p_id_datos_personal,
	    v_direccion, v_barrio, v_telefono, v_cel1, v_cel2, v_correo,
	    v_id_pais, v_id_departamento, v_id_ciudad,
	    v_id_sub_zona, v_id_zona,
	    v_fk_seguridad, now(),
	    v_fk_seguridad, now()
	)
	ON CONFLICT (id_datos_personal)
	DO UPDATE
	    SET direccion = EXCLUDED.direccion,
	        barrio = EXCLUDED.barrio,
	        telefono = EXCLUDED.telefono,
	        celular_uno = EXCLUDED.celular_uno,
	        celular_dos = EXCLUDED.celular_dos,
	        correo = EXCLUDED.correo,
	        id_pais = EXCLUDED.id_pais,
	        id_departamento = EXCLUDED.id_departamento,
	        id_ciudad = EXCLUDED.id_ciudad,
	        id_sub_zona = EXCLUDED.id_sub_zona,
	        id_zona = EXCLUDED.id_zona,
	        fk_seguridad_edicion = EXCLUDED.fk_seguridad_edicion,
	        fecha_edicion = now();
END $procedure$

 LANGUAGE plpgsql
AS $function$
DECLARE
  v_usuario INTEGER;
BEGIN
  v_usuario := COALESCE(current_setting('app.user_id', true)::INTEGER, NULL);

  IF (OLD.fecha_actualizacion IS DISTINCT FROM NEW.fecha_actualizacion)
     OR (OLD.fecha_edicion IS DISTINCT FROM NEW.fecha_edicion)
     OR (OLD.fecha_apertura IS DISTINCT FROM NEW.fecha_apertura) THEN

    INSERT INTO seguridad.log_evento (
      esquema, tabla, id_registro, descripcion_evento, fecha_evento, id_usuario
    )
    VALUES (
      TG_TABLE_SCHEMA,
      TG_TABLE_NAME,
      COALESCE(NEW.id_datos_personal, NEW.id_agencia),
      'Cambio de fechas en ' || TG_TABLE_SCHEMA || '.' || TG_TABLE_NAME,
      CURRENT_TIMESTAMP,
      v_usuario
    );
  END IF;

  RETURN NEW;
END;
$function$
