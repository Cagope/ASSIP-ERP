            dp.documento,
            dp.nombres,

            a.nombre_agencia,

            c.saldo_inicial_cuenta,

            c.fecha_final_cuenta,

            e.tipo_movimiento,

            e.modulo,
            m.descripcion_modulo AS nombre_modulo,
            e.tarjeta,
            e.establecimiento,
            sum(COALESCE(e.valor_debito, 0::double precision) - COALESCE(e.valor_credito, 0::double precision)) OVER (PARTITION BY c.id_agencia, c.id_forma_ahorro, c.codigo_cuenta ORDER BY e.fecha_movimiento, e.hora_movimiento, e.id_extracto_cuenta_ahorro ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS acumulado_mov

             LEFT JOIN depositos.extractos_cuentas_ahorros e ON e.id_cuenta_ahorro = c.id_cuenta_ahorro
             LEFT JOIN depositos.tipo_movimiento tm ON tm.codigo_movimiento = e.tipo_movimiento
             LEFT JOIN depositos.modulos m ON m.codigo_modulo = e.modulo
        )
 SELECT id_cuenta_ahorro,
    codigo_cuenta,

vw_cdat_extractos_total

 SELECT id_extracto_cuenta_cdat,
    id_cuenta_cdat,
    codigo_cdat,
    id_agencia,
    id_datos_personal,
    documento,
    nombre_completo_apellidos,
    nombre_completo_nombres,
    fecha_movimiento,
    hora_movimiento,
    tipo_comprobante,
    numero_comprobante,
    tipo_movimiento,
    descripcion_movimiento,
    valor_debito,
    valor_credito,
    modulo,
    tarjeta,
    fk_seguridad_creacion,
    fecha_creacion,
    sum(valor_credito - valor_debito) OVER (PARTITION BY id_cuenta_cdat ORDER BY fecha_movimiento, hora_movimiento, id_extracto_cuenta_cdat ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS saldo_resultante
   FROM ( SELECT e.id_extracto_cuenta_cdat,
            e.id_cuenta_cdat,
            c.codigo_cdat,
            c.id_agencia,
            c.id_datos_personal,
            hv.documento,
            hv.nombre_completo_apellidos,
            hv.nombre_completo_nombres,
            e.fecha_movimiento,
            e.hora_movimiento,
            e.tipo_comprobante,
            e.numero_comprobante,
            e.tipo_movimiento,
            tm.descripcion AS descripcion_movimiento,
            COALESCE(e.valor_debito, 0::numeric)::numeric(18,2) AS valor_debito,
            COALESCE(e.valor_credito, 0::numeric)::numeric(18,2) AS valor_credito,
            e.modulo,
            e.tarjeta,
            e.fk_seguridad_creacion,
            e.fecha_creacion
           FROM cdat.extractos_cuentas_cdats e
             LEFT JOIN cdat.cuentas_cdats c ON c.id_cuenta_cdat = e.id_cuenta_cdat
             LEFT JOIN cdat.tipos_movimientos_cdats tm ON tm.codigo_movimiento = e.tipo_movimiento
             LEFT JOIN LATERAL ( SELECT hv_1.documento,
                    hv_1.nombre_completo_apellidos,
                    hv_1.nombre_completo_nombres
                   FROM reporting.vw_hoja_vida_general_total_extendida hv_1
                  WHERE hv_1.id_datos_personal = c.id_datos_personal
                 LIMIT 1) hv ON true) x;
